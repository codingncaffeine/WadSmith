version "4.6"

// Uses every wall that opens a door (by tag, a switch, or the area behind with none; Strife's sliding doors too, whose ceiling
// jumps up by their picture's height as they start to slide) or lowers a lift (by tag, or the
// area behind with none) as the player would, one at a time: notes the area's ceiling or floor,
// uses the wall from its front, reads it again 2 seconds later, then waits for the door to close or
// the lift to come back before the next. WadSmith checks the doors and lifts it makes with this.
class WadSmithDoorCheck : StaticEventHandler
{
	Array<int> openers;
	int next;
	int phase;
	int since;
	double closed;
	bool used;
	Sector target;
	Vector3 home;
	double homeAngle;

	// Sliding and swinging doors (polyobjects): their walls' corners as the wall was used, and how far any has moved since.
	Array<int> polyLines;
	Array<double> corners;

	static bool IsPolyDoor(Line l) { return l.special == 7 || l.special == 8; }

	// How long a polyobject door waits open: a negative wait keeps it open.
	static int PolyWait(Line l) { return l.special == 8 ? l.args[4] : l.args[3]; }

	void Snapshot()
	{
		corners.Clear();
		for (int i = 0; i < polyLines.Size(); i++)
		{
			Line p = level.Lines[polyLines[i]];
			corners.Push(p.v1.p.x);
			corners.Push(p.v1.p.y);
		}
	}

	// The farthest any polyobject wall's first corner now lies from where it was at the snapshot.
	double Moved()
	{
		double most = 0;
		for (int i = 0; i < polyLines.Size(); i++)
		{
			Line p = level.Lines[polyLines[i]];
			most = max(most, (p.v1.p - (corners[i * 2], corners[i * 2 + 1])).Length());
		}

		return most;
	}

	// Which way the walls that moved went, on average (a slide's direction, the side a swing went to).
	Vector2 Went()
	{
		Vector2 sum = (0, 0);
		int count = 0;
		for (int i = 0; i < polyLines.Size(); i++)
		{
			Line p = level.Lines[polyLines[i]];
			Vector2 d = p.v1.p - (corners[i * 2], corners[i * 2 + 1]);
			if (d.Length() < 0.5) continue;
			sum += d;
			count++;
		}

		return count > 0 ? sum / count : (0, 0);
	}

	static bool IsTeleport(Line l) { return l.special == 70 || l.special == 71; }

	static bool IsLift(Line l) { return l.special == 62 || l.special == 206; }

	// The area a wall works on: by its tag, or the area behind it when it names none.
	static Sector TargetOf(Line l)
	{
		if (l.args[0] == 0) return l.backsector;
		let it = level.CreateSectorTagIterator(l.args[0]);
		int s = it.Next();
		if (s < 0) return null;
		return level.Sectors[s];
	}

	override void WorldLoaded(WorldEvent e)
	{
		if (e.IsSaveGame || e.IsReopen) return;
		for (int i = 0; i < level.Lines.Size(); i++)
		{
			Line l = level.Lines[i];
			if ((l.special == 11 || l.special == 12 || l.special == 13 || l.special == 14) && (l.args[0] != 0 || l.backsector)) openers.Push(i);
			else if (IsLift(l) && TargetOf(l)) openers.Push(i);
			else if (IsTeleport(l) && (l.activation & SPAC_Cross)) openers.Push(i);
			else if (IsPolyDoor(l)) openers.Push(i);
			if (l.sidedef[0] && (l.sidedef[0].Flags & Side.WALLF_POLYOBJ)) polyLines.Push(i);
		}

		Console.Printf("WSDOOR BEGIN %s %d", level.MapName, openers.Size());
	}

	override void WorldTick()
	{
		// Every key the game has, so locked doors are tried as doors.
		if (level.maptime == 5)
		{
			home = players[consoleplayer].mo.pos;
			homeAngle = players[consoleplayer].mo.angle;
			// With ws_nokeys on (the lock's control), no keys: a locked door must then stay shut.
			let none = CVar.FindCVar("ws_nokeys");
			for (int i = 0; i < AllActorClasses.Size() && !(none && none.GetBool()); i++)
			{
				let k = (class<Key>)(AllActorClasses[i]);
				if (k && k != "Key") players[consoleplayer].mo.GiveInventory(k, 1);
			}
		}

		if (level.maptime < 10 || next > openers.Size()) return;
		if (next == openers.Size())
		{
			Console.Printf("WSDOOR END");
			next++;
			return;
		}

		Line l = level.Lines[openers[next]];
		bool lift = IsLift(l);
		since++;

		// A teleporter: crossed from its front by the player, standing back at the start each time.
		if (IsTeleport(l))
		{
			let mo = players[consoleplayer].mo;
			if (phase == 0)
			{
				mo.SetOrigin(home, false);
				mo.angle = homeAngle;
				used = l.Activate(mo, Line.front, SPAC_Cross);
				phase = 1;
				since = 0;
			}
			else if (since >= 3)
			{
				Console.Printf("WSTELE %d %.1f %.1f %.1f %.1f %d", openers[next], home.x, home.y, mo.pos.x, mo.pos.y, used);
				next++;
				phase = 0;
			}

			return;
		}
		// A sliding or swinging door: used from the front, the polyobjects' corners read 2 seconds later, then back where they were.
		if (IsPolyDoor(l))
		{
			if (phase == 0)
			{
				Snapshot();
				used = l.Activate(players[consoleplayer].mo, Line.front, (l.activation & SPAC_Use) ? SPAC_Use : SPAC_Cross);
				phase = 1;
				since = 0;
			}
			else if (phase == 1 && since >= 70)
			{
				Vector2 went = Went();
				Console.Printf("WSPOLY %d %.1f %d %.1f %.1f", openers[next], Moved(), used, went.x, went.y);
				phase = 2;
			}
			else if (phase == 2 && (PolyWait(l) < 0 || Moved() < 0.5 || since > 700))
			{
				if (PolyWait(l) >= 0) Console.Printf("WSPOLYSHUT %d %.1f %d", openers[next], Moved(), since);
				next++;
				phase = 0;
			}

			return;
		}

		if (phase == 0)
		{
			// Walked-across lift lines are set off by crossing, switches and doors by use.
			target = TargetOf(l);
			if (!target)
			{
				// A switch naming a tag no area has: it can open nothing.
				Console.Printf("WSDOOR %d -1 0 0 0 0", openers[next]);
				next++;
				return;
			}

			closed = lift ? target.CenterFloor() : target.CenterCeiling();
			used = l.Activate(players[consoleplayer].mo, Line.front, (l.activation & SPAC_Use) ? SPAC_Use : SPAC_Cross);
			phase = 1;
			since = 0;
		}
		else if (phase == 1 && since >= 70)
		{
			if (lift) Console.Printf("WSLIFT %d %d %.1f %.1f %d", openers[next], target.Index(), closed, target.CenterFloor(), used);
			else Console.Printf("WSDOOR %d %d %.1f %.1f %.1f %d", openers[next], target.Index(), target.CenterFloor(), closed, target.CenterCeiling(), used);
			phase = 2;
		}
		// Doors that stay open (Door_Open, or locked with no delay) do not close: on to the next wall at once.
		else if (phase == 2 && ((!lift && (l.special == 11 || ((l.special == 13 || l.special == 14) && l.args[2] == 0))) || (lift ? target.CenterFloor() >= closed - 0.5 : target.CenterCeiling() <= closed + 0.5) || since > 700))
		{
			next++;
			phase = 0;
		}
	}
}
